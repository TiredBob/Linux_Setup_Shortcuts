# This document assumes you're using bash, have .bash_aliases in your home folder, and.bashrc is configured to use .bash_aliases.

# Single line script to install Eza, and replace ls and ll in .bash_aliases (needs NerdFonts for Icon Support):
sudo apt update && (sudo apt install -y eza || (case "$(uname -m)" in x86_64) asset=eza_x86_64-unknown-linux-gnu.tar.gz ;; aarch64|arm64) asset=eza_aarch64-unknown-linux-gnu.tar.gz ;; *) echo "Unsupported architecture: $(uname -m)" >&2; exit 1 ;; esac && curl -fsSL -o /tmp/eza.tar.gz "https://github.com/eza-community/eza/releases/latest/download/$asset" && tar -xzf /tmp/eza.tar.gz -C /tmp && sudo install -m755 /tmp/eza /usr/local/bin/eza)) && { grep -qxF "alias ls='eza --icons=auto'" ~/.bash_aliases 2>/dev/null || echo "alias ls='eza --icons=auto'" >> ~/.bash_aliases; grep -qxF "alias ll='eza -lah --icons=auto'" ~/.bash_aliases 2>/dev/null || echo "alias ll='eza -lah --icons=auto'" >> ~/.bash_aliases; } && source ~/.bash_aliases

# Another version without icons:
sudo apt update && (sudo apt install -y eza || (case "$(uname -m)" in x86_64) asset=eza_x86_64-unknown-linux-gnu.tar.gz ;; aarch64|arm64) asset=eza_aarch64-unknown-linux-gnu.tar.gz ;; *) echo "Unsupported architecture: $(uname -m)" >&2; exit 1 ;; esac && curl -fsSL -o /tmp/eza.tar.gz "https://github.com/eza-community/eza/releases/latest/download/$asset" && tar -xzf /tmp/eza.tar.gz -C /tmp && sudo install -m755 /tmp/eza /usr/local/bin/eza)) && { grep -qxF "alias ls='eza'" ~/.bash_aliases 2>/dev/null || echo "alias ls='eza'" >> ~/.bash_aliases; grep -qxF "alias ll='eza -lah'" ~/.bash_aliases 2>/dev/null || echo "alias ll='eza -lah'" >> ~/.bash_aliases; } && source ~/.bash_aliases

# duf as df (only showing local filesystems):
(command -v duf >/dev/null 2>&1 || { sudo apt update && sudo apt install -y duf; }) && { grep -qxF "alias df='duf --only local'" ~/.bash_aliases 2>/dev/null || echo "alias df='duf --only local'" >> ~/.bash_aliases; } && source ~/.bash_aliases

# And duf as df using standard duf:
(command -v duf >/dev/null 2>&1 || { sudo apt update && sudo apt install -y duf; }) && { grep -qxF "alias df='duf'" ~/.bash_aliases 2>/dev/null || echo "alias df='duf'" >> ~/.bash_aliases; } && source ~/.bash_aliases

# Alias to upgrade installed packages (run upgrade from the command line) works with apt/dnf/pacman:
grep -qF '# upgrade function: apt/dnf/pacman' ~/.bash_aliases 2>/dev/null || cat >> ~/.bash_aliases <<'EOF'
# upgrade function: apt/dnf/pacman
upgrade() {
  if command -v apt >/dev/null 2>&1; then
    sudo apt update && sudo apt full-upgrade -y && sudo apt autoremove -y
  elif command -v dnf >/dev/null 2>&1; then
    sudo dnf upgrade --refresh -y && sudo dnf autoremove -y
  elif command -v pacman >/dev/null 2>&1; then
    sudo pacman -Syu --noconfirm
  else
    echo "No supported package manager found (apt, dnf, or pacman)." >&2
    return 1
  fi
}
EOF
source ~/.bash_aliases


# If you don't want to use the upgrade command, and prefer auto updates (Ubuntu/Deb based):
sudo apt install unattended-upgrades && sudo dpkg-reconfigure unattended-upgrades
